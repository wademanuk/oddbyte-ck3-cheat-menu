twcm.0001 = {
    type = character_event
    title = twcm_1_title
    desc = twcm_1_desc
    theme = stewardship
    left_portrait = scope:twcm_target
    trigger = { is_ai = no }
    soundeffect = none
    immediate = {
        if = {
            limit = { has_variable = twcm_ui_other exists = var:twcm_ui_other var:twcm_ui_other ?= { is_alive = yes } }
            var:twcm_ui_other = { save_scope_as = twcm_other }
        }
        else = { save_scope_as = twcm_other }
        if = {
            limit = { NOT = { exists = scope:twcm_target } }
            save_scope_as = twcm_target
        }
        if = {
            limit = { scope:twcm_target = { is_alive = no } }
            save_scope_as = twcm_target
        }
    }
    option = {
        name = twcm_resources
        hidden_effect = { trigger_event = { id = twcm.0010 } }
    }
    option = {
        name = twcm_stress
        scope:twcm_target = {
        add_stress = -1000
        }
        hidden_effect = { trigger_event = { id = twcm.0001 } }
    }
    option = {
        name = twcm_skills
        hidden_effect = { trigger_event = { id = twcm.0020 } }
    }
    option = {
        name = twcm_traits
        hidden_effect = { trigger_event = { id = twcm.0030 } }
    }
    option = {
        name = twcm_construction
        hidden_effect = { trigger_event = { id = twcm.0040 } }
    }
    option = {
        name = twcm_more

}
